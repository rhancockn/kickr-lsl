# FTMS Target Slew Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** The FTMS service ramps to each new target at a per-client slew rate (default 10 W/s), sending interpolated setpoints every `slew_update_interval_s` (default 10 s); slew 0 keeps today's behaviour. Exposed via `KickrClient.set_slew_rate`, `wahoo-ftms set --slew`, and the dashboard.

**Architecture:** A pure `ftms/slew.py` `Ramp` computes interpolated values and update times. `Controller` owns the ramp: `request_target` / `set_slew_rate` (re)start it from the current interpolated value, and the sender loop either runs today's rate-limited path (no ramp) or a ramp step (sleep until the next update or a wake-up, send the clamped value). Server adds a `slew` command and resets the rate on client disconnect. Client, CLI, dashboard worker/app and launcher follow.

**Tech Stack:** Python ≥ 3.10, asyncio, Tkinter, pytest.

**Spec:** `docs/superpowers/specs/2026-10-22-ftms-slew-design.md`

```python
"""Target-power ramp for slewed FTMS targets (spec: FTMS target slew). Pure: no I/O, no clocks."""
from __future__ import annotations

import math
from dataclasses import dataclass

MAX_SLEW_W_PER_S = 1000.0
_GRID_EPS = 1e-9  # a time on (or a rounding error below) an update grid point counts as that point


@dataclass(frozen=True)
class Ramp:
    """Linear move from start_w to target_w at rate_w_per_s, starting at t0.

    Updates are due at t0 + k * interval_s (k >= 1) and finally at end_t.
    """

    start_w: float
    target_w: int
    rate_w_per_s: float
    t0: float
    interval_s: float

    def __post_init__(self) -> None:
        if not (math.isfinite(self.rate_w_per_s) and self.rate_w_per_s > 0):
            raise ValueError(f"rate_w_per_s must be a finite number > 0, got {self.rate_w_per_s!r}")
        if not (math.isfinite(self.interval_s) and self.interval_s > 0):
            raise ValueError(f"interval_s must be a finite number > 0, got {self.interval_s!r}")

    @property
    def end_t(self) -> float:
        return self.t0 + abs(self.target_w - self.start_w) / self.rate_w_per_s

    def value_at(self, t: float) -> float:
        if t >= self.end_t:
            return float(self.target_w)
        if t <= self.t0:
            return float(self.start_w)
        step = self.rate_w_per_s * (t - self.t0)
        return self.start_w + step if self.target_w > self.start_w else self.start_w - step

    def next_update_at(self, after_t: float) -> float:
        """The first update time strictly after ``after_t``, capped at end_t."""
        k = max(1, math.floor((after_t - self.t0) / self.interval_s + _GRID_EPS) + 1)
        return min(self.t0 + k * self.interval_s, self.end_t)
```

        if rate == 0:
            self._ramp = None  # the sender sends the pending target directly
        elif self._ramp is not None:
            self._start_ramp(self.state.requested_w)
        self._wake.set()
        return rate

        """Back to the configured rate, no ramp (the client that set the rate has gone)."""
        self.slew_w_per_s = self._default_slew
        self._ramp = None

        return self._ramp.value_at(now) if self._ramp is not None else float(self._sent_w)

        start = self._setpoint(now)
        if start == target:
            self._ramp = None
            return
        self._ramp = Ramp(start, target, self.slew_w_per_s, now, self._slew_interval)

    async def _sender(self) -> None:
        while True:
            if self._ramp is not None:
                await self._ramp_step()
                continue
            await self._wake.wait()
            self._wake.clear()
            if self._ramp is not None or not self.pending:
                continue
            wait = self._last_send + self._min_interval - self._now()
            if wait > 0:
                await self._sleep(wait)
            if self._ramp is not None or not self.pending:
                continue  # a ramp started, the link was lost, or the latest request equals what is confirmed
            target = self.state.requested_w  # read after the wait: latest wins
            if self._requests_since_send > 1:
                log.debug("coalesced %d target(s); sending %d W", self._requests_since_send - 1, target)
            self._requests_since_send = 0
            self._last_send = self._now()
            await self._send(target)

    async def _ramp_step(self) -> None:
        """Wait for the ramp's next update (or a new request / rate change), then send it."""
        ramp = self._ramp
        assert ramp is not None
        self._wake.clear()
        due = max(ramp.next_update_at(self._last_update_t), self._last_send + self._min_interval)
        delay = due - self._now()
        if delay > 0 and await self._sleep_or_wake(delay):
            return  # woken: the ramp or the rate changed, recompute
        if self._ramp is not ramp or self.state.link != "ready":
            return
        now = self._now()
        self._last_update_t = now
        final = now >= ramp.end_t - RAMP_END_TOLERANCE_S
        value = ramp.target_w if final else self.clamp(round(ramp.value_at(now)))
        if final:
            self._ramp = None
        if value == self._sent_w and self.state.confirmed_w == value:
            return  # rounds to what the trainer already holds
        self._requests_since_send = 0
        self._last_send = now
        await self._send(value)

    async def _sleep_or_wake(self, seconds: float) -> bool:
        """Sleep up to ``seconds`` with the injected sleep; True if woken early by _wake."""
        sleeper = asyncio.ensure_future(self._sleep(seconds))
        waker = asyncio.ensure_future(self._wake.wait())
        try:
            done, _ = await asyncio.wait({sleeper, waker}, return_when=asyncio.FIRST_COMPLETED)
        finally:
            for task in (sleeper, waker):
                task.cancel()
            await asyncio.gather(sleeper, waker, return_exceptions=True)
        if waker in done:
            self._wake.clear()
            return True
        return False
